import { PrismaClient, Prisma } from '@prisma/client';
import { randomUUID } from 'crypto';

const prisma = new PrismaClient();

const EGYPTIAN_FIRST_NAMES = ['Ahmed', 'Mohamed', 'Fatma', 'Sara', 'Omar', 'Youssef', 'Nour', 'Hassan', 'Mariam', 'Khaled'];
const EGYPTIAN_LAST_NAMES = ['Hassan', 'Ali', 'Mahmoud', 'Ibrahim', 'El-Sayed', 'Reda', 'Khalil', 'Mostafa', 'Salem', 'Zaki'];

function seededRandom(seed: number) {
  let x = Math.sin(seed) * 10000;
  return x - Math.floor(x);
}

export async function runAdvancedSeeding(opts: {
  scenario: 'hurghada-red-sea-compound';
  pastDays?: number;
  totalScans?: number;
  incidentRate?: number;
  organizationId?: string;
  randomSeed?: number;
}) {
  const {
    scenario = 'hurghada-red-sea-compound',
    pastDays = 45,
    totalScans = 5000,
    incidentRate = 0.12,
    organizationId,
    randomSeed = Date.now(),
  } = opts;

  const rand = (n: number) => Math.floor(seededRandom(randomSeed + n) * 1000) % n;

  console.log(`🚀 Starting Hurghada Red Sea emulation (seed=${randomSeed})...`);

  // 1. Organization (demo or existing)
  let orgId = organizationId;
  if (!orgId) {
    const org = await prisma.organization.create({
      data: {
        name: 'RedSea Paradise Compound - Hurghada',
        plan: 'ENTERPRISE',
        settings: { retentionDays: 90, locationEnforcement: true },
      },
    });
    orgId = org.id;
  }

  // 2. Project
  const project = await prisma.project.create({
    data: {
      name: 'RedSea Paradise',
      organizationId: orgId!,
      description: 'Luxury beach compound - Hurghada, Red Sea, Egypt',
    },
  });

  // 3. Gates (realistic for compound)
  const gates = await prisma.gate.createMany({
    data: [
      { id: randomUUID(), name: 'Main Gate', organizationId: orgId!, projectId: project.id, latitude: 27.2578, longitude: 33.8119, radiusMeters: 80 },
      { id: randomUUID(), name: 'Beach Gate', organizationId: orgId!, projectId: project.id, latitude: 27.2551, longitude: 33.8152, radiusMeters: 60 },
      { id: randomUUID(), name: 'Service Gate', organizationId: orgId!, projectId: project.id, latitude: 27.2603, longitude: 33.8087, radiusMeters: 50 },
    ],
    skipDuplicates: true,
  });

  // 4. Units + Owners (high-volume)
  const unitsData: Prisma.UnitCreateManyInput[] = [];
  const contactsData: Prisma.ContactCreateManyInput[] = [];
  for (let i = 1; i <= 1200; i++) {
    const ownerName = `${EGYPTIAN_FIRST_NAMES[rand(EGYPTIAN_FIRST_NAMES.length)]} ${EGYPTIAN_LAST_NAMES[rand(EGYPTIAN_LAST_NAMES.length)]}`;
    const unitNumber = i <= 200 ? `Villa ${i}` : `Tower A-${i - 200}`;

    unitsData.push({
      unitNumber,
      unitType: i % 7 === 0 ? 'VILLA' : i % 3 === 0 ? 'PENTHOUSE' : 'THREE_BEDROOM',
      organizationId: orgId!,
      projectId: project.id,
      isActive: true,
    });

    contactsData.push({
      name: ownerName,
      phone: `+2010${String(1000000 + rand(9000000))}`,
      email: `${ownerName.toLowerCase().replace(' ', '.')}.rs@hurghada.eg`,
      organizationId: orgId!,
      isResident: true,
    });
  }
  await prisma.unit.createMany({ data: unitsData, skipDuplicates: true });
  await prisma.contact.createMany({ data: contactsData, skipDuplicates: true });

  // 5. Emulate Resident Visitor QR Creation (real flow)
  const visitorQRs: Prisma.VisitorQRCreateManyInput[] = [];
  const qrCodes: Prisma.QRCodeCreateManyInput[] = [];
  const unitIds = await prisma.unit.findMany({ where: { organizationId: orgId! }, select: { id: true } });

  for (let i = 0; i < 800; i++) {
    const qrId = randomUUID();
    const unit = unitIds[rand(unitIds.length)];

    qrCodes.push({
      id: qrId,
      code: `GF-HUR-${String(10000 + i).padStart(5, '0')}`,
      type: 'VISITOR',
      organizationId: orgId!,
      projectId: project.id,
      expiresAt: new Date(Date.now() - rand(pastDays * 86400000 * 0.7)), // mostly still valid
      maxUses: rand(5) + 1,
    });

    visitorQRs.push({
      id: randomUUID(),
      qrCodeId: qrId,
      unitId: unit.id,
      visitorName: `${EGYPTIAN_FIRST_NAMES[rand(10)]} ${EGYPTIAN_LAST_NAMES[rand(10)]}`,
      visitorPhone: `+2012${String(1000000 + rand(9000000))}`,
      isOpenQR: rand(10) > 7,
      organizationId: orgId!,
    });
  }

  await prisma.qRCode.createMany({ data: qrCodes, skipDuplicates: true });
  await prisma.visitorQR.createMany({ data: visitorQRs, skipDuplicates: true });

  // 6. Rush-Hour Gate Traffic Emulation (ScanLogs)
  console.log(`🌊 Generating ${totalScans} realistic scans with Rush Hour clustering...`);
  const scanLogs: Prisma.ScanLogCreateManyInput[] = [];
  const baseDate = new Date();
  baseDate.setDate(baseDate.getDate() - pastDays);

  for (let i = 0; i < totalScans; i++) {
    const daysAgo = rand(pastDays);
    const hour = rand(24);
    let scanDate = new Date(baseDate);
    scanDate.setDate(scanDate.getDate() + daysAgo);

    // Rush hour clustering (realistic for Egyptian compounds)
    if (hour >= 7 && hour <= 10) scanDate.setHours(8 + rand(3), rand(60)); // morning entry
    else if (hour >= 17 && hour <= 21) scanDate.setHours(18 + rand(3), rand(60)); // evening return
    else scanDate.setHours(hour, rand(60));

    const statusRoll = rand(100);
    let status: 'SUCCESS' | 'DENIED' | 'EXPIRED' = 'SUCCESS';
    if (statusRoll < incidentRate * 100) status = 'DENIED';
    else if (statusRoll < 25) status = 'EXPIRED';

    scanLogs.push({
      id: randomUUID(),
      qrCodeId: qrCodes[rand(qrCodes.length)].id!,
      gateId: gates[rand(3)].id!,
      organizationId: orgId!,
      scannedAt: scanDate,
      status,
      operatorId: null, // can be extended later
      scanUuid: randomUUID(),
    });
  }

  // Batch insert (as required by IDEA_advanced_seeding_v2)
  const BATCH_SIZE = 500;
  for (let i = 0; i < scanLogs.length; i += BATCH_SIZE) {
    await prisma.scanLog.createMany({
      data: scanLogs.slice(i, i + BATCH_SIZE),
      skipDuplicates: true,
    });
  }

  // Optional: Incidents for watchlist/security testing
  if (incidentRate > 0) {
    // ... (simple incident seeding omitted for brevity — follows same pattern)
  }

  console.log(`✅ Hurghada Red Sea emulation complete! ${totalScans} scans seeded over ${pastDays} days.`);
  return { organizationId: orgId, totalScans, pastDays };
}

// Helper: Gaussian approximation around peak (reproducible)
function gaussianSample(mean: number, stdDev: number, randFn: (offset: number) => number, offset: number): number {
  // Box-Muller transform (simple 2-sample version)
  const u1 = randFn(offset);
  const u2 = randFn(offset + 1);
  const z0 = Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
  return Math.max(0, Math.min(23, Math.floor(mean + z0 * stdDev)));
}

// Main expanded Rush Hour generator
function generateRushHourScanDate(
  baseDate: Date,
  pastDays: number,
  scenario: Scenario,
  randomSeed: number,
  scanIndex: number,
  randFn: (n: number, idx: number) => number
): Date {
  const config = SCENARIO_CONFIG[scenario];
  const dayOffset = randFn(pastDays, scanIndex);
  const scanDate = new Date(baseDate);
  scanDate.setDate(scanDate.getDate() + dayOffset);

  const isWeekend = scanDate.getDay() === 5 || scanDate.getDay() === 6; // Friday/Saturday (MENA)

  // 70% chance of rush-hour scan, 30% baseline random for valleys
  if (seededRandom(randomSeed + scanIndex + 500) < 0.3) {
    // Baseline non-rush (spread throughout day)
    scanDate.setHours(randFn(24, scanIndex + 600), randFn(60, scanIndex + 700));
    return scanDate;
  }

  // Choose rush period with weighted probability
  const periods = config.rushPeriods;
  const weights = periods.map(p => 
    p.intensity * (isWeekend ? p.weekendWeight : p.weekdayWeight)
  );
  let totalWeight = weights.reduce((a, b) => a + b, 0);
  let r = seededRandom(randomSeed + scanIndex + 800) * totalWeight;
  const chosenPeriod = periods.find((p, i) => {
    r -= weights[i];
    return r <= 0;
  })!;

  // Gaussian clustering inside the chosen window
  const stdDev = (chosenPeriod.endHour - chosenPeriod.startHour) / 3.5; // natural spread
  const hour = gaussianSample(
    chosenPeriod.peakHour,
    stdDev,
    (offset) => seededRandom(randomSeed + scanIndex * 10 + offset),
    scanIndex * 10 + 900
  );

  // Clamp to period bounds
  const finalHour = Math.max(chosenPeriod.startHour, Math.min(chosenPeriod.endHour, hour));

  scanDate.setHours(finalHour, randFn(60, scanIndex + 1000));
  return scanDate;
}

TypeScriptfunction getBedrooms(scenario: Scenario, counter: number, isVilla: boolean): number {

3. New generateUnitId Function
TypeScriptfunction generateUnitId(
  phaseIndex: number,
  buildingCode: string,
  floorNumber: number,
  unitNumberInBuilding: number,
  formatType: UnitIdFormatType,
  phaseName: string
): string {
  const phaseCode = `P${phaseIndex + 1}`;
  const floorCode = `F${String(floorNumber).padStart(2, '0')}`;
  const unitCode = String(unitNumberInBuilding).padStart(3, '0');

  switch (formatType) {
    case 'compact':
      return `${phaseCode}-${buildingCode}-${floorCode}-${unitCode}`;

    case 'building-first':
      return `${buildingCode}-${floorCode}-${unitCode}`;

    case 'simple':
      return `${buildingCode}-${unitCode}`;

    case 'location':
      const locationPrefix = phaseName.includes('Seafront') ? 'SF' 
                           : phaseName.includes('Garden') ? 'GD' 
                           : phaseName.includes('Marina') ? 'MR' : 'P';
      return `${locationPrefix}-${buildingCode}-${String(floorNumber).padStart(2, '0')}-${unitCode}`;

    case 'descriptive':
      const shortPhase = phaseName.split(' - ')[0].replace('Phase ', 'P');
      return `${shortPhase}-${buildingCode}-${floorCode}-${unitCode}`;

    case 'global':
      return `GF-${phaseCode}-${buildingCode}-${floorCode}-${unitCode}`;

    default:
      return `${phaseCode}-${buildingCode}-${floorCode}-${unitCode}`;
  }
}
